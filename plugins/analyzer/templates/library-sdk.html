Usage Examples

# Initialize the SDK client
from sdk import create_client
from sdk.exceptions import SDKError, AuthError

client = create_client(
    api_key="sk-xxxxxxxxxxxx",
    timeout=30,
    retries=3,
)

# List all projects
projects = client.projects.list(limit=10)
for project in projects:
    print(project.name, project.status)

# Create a deployment with error handling
try:
    deployment = client.deployments.create(
        project_id="proj-abc123",
        image="myapp:latest",
    )
    print(f"Deployed: {deployment.id}")
except AuthError as exc:
    print(f"Auth failed: {exc}")
except SDKError as exc:
    print(f"SDK error: {exc}")
